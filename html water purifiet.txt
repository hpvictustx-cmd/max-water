/* =========================================================
   MATERIAL HELPERS
========================================================= */

function mat(color,metalness=.55,roughness=.35){

    return new THREE.MeshStandardMaterial({
        color,
        metalness,
        roughness
    });
}


function transparentMat(
    color,
    opacity=.25
){

    return new THREE.MeshPhysicalMaterial({
        color,
        metalness:.15,
        roughness:.15,
        transparent:true,
        opacity,
        transmission:.1
    });
}


/* =========================================================
   CYLINDER BETWEEN TWO POINTS
========================================================= */

function cylinderBetween(
    a,
    b,
    radius,
    material
){

    const direction = new THREE.Vector3()
        .subVectors(b,a);

    const length = direction.length();

    const geometry =
        new THREE.CylinderGeometry(
            radius,
            radius,
            length,
            18
        );

    const mesh =
        new THREE.Mesh(
            geometry,
            material
        );

    mesh.position
        .copy(a)
        .add(b)
        .multiplyScalar(.5);

    mesh.quaternion.setFromUnitVectors(
        new THREE.Vector3(0,1,0),
        direction.normalize()
    );

    mesh.castShadow = true;

    mesh.receiveShadow = true;

    systemGroup.add(mesh);

    return mesh;
}


/* =========================================================
   FRAME
========================================================= */

function createFrame(){

    const metal = mat(DARK_METAL);

    const posts = [

        [-15,0,0,-15,18,0],
        [15,0,0,15,18,0],

        [-15,0,-12,-15,18,-12],
        [15,0,-12,15,18,-12]

    ];

    posts.forEach(p=>{
        cylinderBetween(
            new THREE.Vector3(p[0],p[1],p[2]),
            new THREE.Vector3(p[3],p[4],p[5]),
            .32,
            metal
        );
    });


    /* TOP FRAME */

    cylinderBetween(
        new THREE.Vector3(-15,18,0),
        new THREE.Vector3(15,18,0),
        .28,
        metal
    );

    cylinderBetween(
        new THREE.Vector3(-15,18,-12),
        new THREE.Vector3(15,18,-12),
        .28,
        metal
    );


    /* CROSS BRACES */

    cylinderBetween(
        new THREE.Vector3(-15,1,-12),
        new THREE.Vector3(15,17,-12),
        .15,
        metal
    );

    cylinderBetween(
        new THREE.Vector3(15,1,-12),
        new THREE.Vector3(-15,17,-12),
        .15,
        metal
    );
}


/* =========================================================
   PIPE SYSTEM
========================================================= */

function createPipes(){

    const rawPipe = mat(BLUE,.45,.25);

    const cleanPipe = mat(WATER,.35,.2);


    /* RAW WATER */

    cylinderBetween(
        new THREE.Vector3(-25,8,2),
        new THREE.Vector3(-17,8,2),
        .45,
        rawPipe
    );

    cylinderBetween(
        new THREE.Vector3(-17,8,2),
        new THREE.Vector3(-17,12,2),
        .45,
        rawPipe
    );


    /* TESTER TO PREFILTER */

    cylinderBetween(
        new THREE.Vector3(-17,12,2),
        new THREE.Vector3(-10,12,2),
        .4,
        rawPipe
    );


    /* PREFILTER TO PURIFIER */

    cylinderBetween(
        new THREE.Vector3(-7,12,2),
        new THREE.Vector3(-3,12,2),
        .4,
        rawPipe
    );


    /* PURIFIER TO TURBINE */

    cylinderBetween(
        new THREE.Vector3(3,12,2),
        new THREE.Vector3(8,12,2),
        .4,
        cleanPipe
    );


    /* DOWN TO TURBINE */

    cylinderBetween(
        new THREE.Vector3(8,12,2),
        new THREE.Vector3(8,7,2),
        .4,
        cleanPipe
    );


    /* TURBINE TO HOUSE */

    cylinderBetween(
        new THREE.Vector3(8,5,2),
        new THREE.Vector3(18,5,2),
        .42,
        cleanPipe
    );


    cylinderBetween(
        new THREE.Vector3(18,5,2),
        new THREE.Vector3(25,5,2),
        .42,
        cleanPipe
    );
}

/* =========================================================
   UPDATE LABEL POSITIONS
========================================================= */

function updateLabels(){

    if(!renderer) return;

    const rect =
        renderer.domElement.getBoundingClientRect();


    labels.forEach(label=>{

        label.element.style.display =
            labelsVisible ? "block" : "none";


        if(!labelsVisible) return;


        const world =
            label.position.clone();

        systemGroup.localToWorld(world);

        world.project(camera);


        const x =
            (world.x*.5+.5)*rect.width;

        const y =
            (-world.y*.5+.5)*rect.height;


        label.element.style.left =
            x+"px";

        label.element.style.top =
            y+"px";
    });
}


/* =========================================================
   WATER PARTICLE ANIMATION
========================================================= */

function updateWaterParticles(delta){

    if(!running || flow<=0){

        waterParticles.forEach(p=>{
            p.visible=false;
        });

        return;
    }


    waterParticles.forEach((p,i)=>{

        p.visible=true;

        let progress =
            (clock.elapsedTime*
            .08*
            (flow/8.5)*
            p.userData.speed+
            p.userData.offset)%1;


        let x;
        let y;
        let z=2;


        /* RAW PIPE */

        if(progress<.20){

            const t=progress/.20;

            x=-25+8*t;

            y=8;

        }

        /* TESTER */

        else if(progress<.32){

            const t=(progress-.20)/.12;

            x=-17;

            y=8+4*t;

        }

        /* TESTER TO FILTER */

        else if(progress<.48){

            const t=(progress-.32)/.16;

            x=-17+12*t;

            y=12;

        }

        /* PURIFIER */

        else if(progress<.70){

            const t=(progress-.48)/.22;

            x=-5+8*t;

            y=12;

        }

        /* DOWN PIPE */

        else if(progress<.82){

            const t=(progress-.70)/.12;

            x=8;

            y=12-7*t;

        }

        /* TURBINE TO HOUSE */

        else{

            const t=(progress-.82)/.18;

            x=8+17*t;

            y=5;
        }


        p.position.set(
            x,
            y,
            z+(Math.random()-.5)*.5
        );

    });
}


/* =========================================================
   UPDATE TURBINE
========================================================= */

function updateMachines(delta){

    if(
        running &&
        generatorOn &&
        flow>0
    ){

        const speed =
            flow/8.5;

        turbineRotor.rotation.x +=
            delta*12*speed;

        generatorRotor.rotation.x +=
            delta*9*speed;

    }


    /* POWER */

    let power=0;

    if(
        running &&
        generatorOn &&
        flow>0
    ){

        power =
            12.4*(flow/8.5);
    }


    /* BATTERY */

    if(power>0){

        battery +=
            delta*.35;

        energy +=
            power*delta/3600000;

        if(battery>100)
            battery=100;
    }


    updateDashboard(power);
}

    let turbidity=2.4;


    if(purifierOn){

        tds =
            248 -
            (flow/12)*70;

        ph =
            7.15+
            Math.sin(clock.elapsedTime*.3)*.05;

        turbidity =
            2.4-
            (flow/12)*.7;

    document.getElementById(
        "tdsDisplay"

        ph.toFixed(2);

function resize(){

    const container =
        document.getElementById(
            "threeContainer"
        );

    if(!renderer || !camera)
        return;


    const width =
        container.clientWidth;

    const height =
        container.clientHeight;


    camera.aspect =
        width/height;

    camera.updateProjectionMatrix();


    renderer.setSize(
        width,
        height
    );
}
